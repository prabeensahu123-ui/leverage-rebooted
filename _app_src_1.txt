        stt["open_trades"] = ot
        stt.setdefault("history", [])
        return stt
    except Exception:
        return {"open_trades": [], "history": []}

def rsi_from_closes(closes, period=14):
    """Wilder RSI (TradingView-compatible). Needs oldest→newest closes."""
    if closes is None or len(closes) < period + 1:
        return None
    c = np.asarray(closes, dtype=float)
    d = np.diff(c)
    gains = np.where(d > 0, d, 0.0)
    losses = np.where(d < 0, -d, 0.0)
    avg_gain = float(np.mean(gains[:period]))
    avg_loss = float(np.mean(losses[:period]))
    for i in range(period, len(gains)):
        avg_gain = (avg_gain * (period - 1) + float(gains[i])) / period
        avg_loss = (avg_loss * (period - 1) + float(losses[i])) / period
    if avg_loss < 1e-12:
        return 100.0 if avg_gain > 0 else 50.0
    rs = avg_gain / avg_loss
    return float(100.0 - (100.0 / (1.0 + rs)))

def tf_price_change(candles):
    if not candles or len(candles) < 2:
        return None
    a = float(candles[-2].get("close") or 0)
    b = float(candles[-1].get("close") or 0)
    if a <= 0:
        return None
    return float((b - a) / a * 100.0)

def _macd_bias(closes):
    if closes is None or len(closes) < 35:
        return "—"
    c = np.asarray(closes, dtype=float)
    def ema_s(arr, period):
        k = 2.0 / (period + 1.0)
        out = np.empty(len(arr))
        out[0] = arr[0]
        for i in range(1, len(arr)):
            out[i] = arr[i] * k + out[i - 1] * (1 - k)
        return out
    ef = ema_s(c, 12)
    es = ema_s(c, 26)
    macd = ef - es
    sig = float(np.mean(macd[-9:]))
    return "bull" if macd[-1] > sig else "bear"

def _trend_bias(closes):
    if closes is None or len(closes) < 20:
        return "—"
    c = np.asarray(closes, dtype=float)
    px = float(c[-1])
    sma20 = float(np.mean(c[-20:]))
    sma50 = float(np.mean(c[-50:])) if len(c) >= 50 else float(np.mean(c))
    if px > sma20 > sma50:
        return "bullish"
    if px < sma20 < sma50:
        return "bearish"
    return "mixed"

def _tech_score_parts(chg, rsi, trend, macd):
    score = 0
    if chg is not None:
        if chg > 0.15:
            score += 1
        elif chg < -0.15:
            score -= 1
    if rsi is not None:
        if rsi < 35:
            score += 1
        elif rsi > 65:
            score -= 1
    if trend == "bullish":
        score += 1
    elif trend == "bearish":
        score -= 1
    if macd == "bull":
        score += 1
    elif macd == "bear":
        score -= 1
    return score

def _tech_decision(score):
    if score >= 2:
        return "LONG"
    if score <= -2:
        return "SHORT"
    return "HOLD"

def _confidence_pct(score):
    return int(28 + abs(int(score)) * 15.5)

def _color_side(side):
    if side == "LONG":
        return "🟢 LONG"
    if side == "SHORT":
        return "🔴 SHORT"
    if side == "HOLD":
        return "⚪ HOLD"
    return side or "—"

def _color_reconcile(rec):
    if rec == "AGREE":
        return "✅ AGREE"
    if rec == "CONFLICT":
        return "❌ CONFLICT"
    if rec == "WAIT":
        return "⏸️ WAIT"
    return rec or "—"

def build_tf_row(symbol, tf_key, conf_result=None):
    cfg = CONF_TFS[tf_key]
    candles = fetch_candles(symbol, cfg["resolution"], cfg["days"])
    closes = [float(x["close"]) for x in candles] if candles else []
    chg = tf_price_change(candles)
    rsi = rsi_from_closes(closes) if closes else None
    trend = _trend_bias(closes)
    macd = _macd_bias(closes)
