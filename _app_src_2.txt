    conf_side_display = "— (score 6-pt first)"
    if conf_result:
        conf_side = conf_result.get("side") or "HOLD"
        verdict = conf_result.get("verdict") or conf_side
        ls = conf_result.get("long_score") or 0
        ss = conf_result.get("short_score") or 0
        scored_tf = conf_result.get("data_window")
        conf_side_display = "%s %s · L%d/S%d" % (_color_side(conf_side), verdict, ls, ss)
        if scored_tf and scored_tf != tf_key:
            conf_side_display = "%s (scored on %s)" % (conf_side_display, scored_tf)
    if not conf_result or conf_side is None:
        reconcile = "—"
    elif conf_side == "HOLD" or tech == "HOLD":
        reconcile = "WAIT"

        "Confidence %": conf_pct,
        "Confluence / Liq": conf_side_display,
        "Reconcile": _color_reconcile(reconcile),
    }


def _atr_pct_from_candles(candles, period=14):
    if not candles or len(candles) < period + 2:
        return None
    trs = []
    for i in range(1, len(candles)):
        h = float(candles[i].get("high") or 0)
        l = float(candles[i].get("low") or 0)
        pc = float(candles[i - 1].get("close") or 0)
        if h <= 0 or l <= 0 or pc <= 0:
            continue
        trs.append(max(h - l, abs(h - pc), abs(l - pc)))
    if len(trs) < period:
        return None
    atr = float(sum(trs[-period:]) / period)
    last = float(candles[-1].get("close") or 0)
    if last <= 0:
        return None
    return atr / last * 100.0
