            sl = entry * (1.0 - default_dist)
            sl_source = "ATR / default BELOW"
        stop_dist = (entry - sl) / entry
        tp = entry * (1.0 + stop_dist * rr)
    else:
        liq = cr.get("liq_point_above")
        if liq and float(liq) > 0 and float(liq) > entry:
            cand = float(liq) * 1.002
            dist = (cand - entry) / entry
            if 0.004 <= dist <= 0.08:
                sl = cand
                sl_source = "liq peak ABOVE"
        if sl is None:
            sl = entry * (1.0 + default_dist)
            sl_source = "ATR / default ABOVE"
        stop_dist = (sl - entry) / entry
        tp = entry * (1.0 - stop_dist * rr)

    risk_usd = float(equity) * (float(risk_pct) / 100.0)
    if stop_dist < 1e-8:
        out["notes"].append("Stop too close")
        return out
    notional = risk_usd / stop_dist
    margin = notional / max(float(leverage), 1.0)
    if margin > float(equity):
        margin = float(equity)
        notional = margin * max(float(leverage), 1.0)
        risk_usd = notional * stop_dist
        out["notes"].append("Margin capped at equity")

    out.update({
        "ok": True,
        "entry": round(entry, 2),
        "stop": round(sl, 2),
        "tp": round(tp, 2),
        "stop_dist_pct": round(stop_dist * 100.0, 3),
        "risk_usd": round(risk_usd, 2),
        "notional": round(notional, 2),
        "margin": round(margin, 2),
        "sl_source": sl_source,
        "rr": rr,
    })
    return out
